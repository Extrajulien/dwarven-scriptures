      if (matchTimer) {
        matchTimer.textContent = '00:' + formatted + ' REMAINING';
      }

    input.addEventListener('input', (e) => {
      const len = input.value.length;

      const currentWpm = Math.min(115, Math.max(65, 80 + Math.floor(len / 4)));
      if (headerWpm) headerWpm.textContent = currentWpm + ' WPM';
      if (laneUserWpm) laneUserWpm.textContent = currentWpm + ' WPM';
      if (hudWpm) hudWpm.textContent = currentWpm + ' WPM';
